\passo{2.6}{Criar o arquivo \texttt{js/app.js} do Curso 2 (Filtro em 32 linhas)}

Na oficina, o participante cadastra novos pontos de coleta da cidade no array \texttt{LOCAIS\_COLETA} e compreende a função \texttt{filtrarEcopontos(categoria)}:

\begin{boxCodigoJS}[starter-kit-painel-verde/js/app.js]
// BLOCO 2: Adicione novos ecopontos ou pontos de entrega voluntaria de Garopaba
const LOCAIS_COLETA = [
  {
    nome: "Ecoponto Central — Secretaria de Meio Ambiente",
    bairro: "Centro",
    categoria: "Eletronicos",
    horario: "Segunda a Sexta, 08h as 17h",
    mapaUrl: "https://maps.google.com/?q=Garopaba+Centro"
  },
  {
    nome: "Ponto Verde IFSC Campus Garopaba",
    bairro: "Campo D'Una",
    categoria: "Reciclaveis",
    horario: "Segunda a Sexta, 08h as 22h",
    mapaUrl: "https://maps.google.com/?q=IFSC+Garopaba"
  },
  {
    nome: "Coleta de Vidro e Oleo — Associacao Comunitaria",
    bairro: "Ferrugem",
    categoria: "Vidro",
    horario: "Sabados, 09h as 13h",
    mapaUrl: "https://maps.google.com/?q=Praia+da+Ferrugem+Garopaba"
  }
];

function filtrarEcopontos(categoriaEscolhida) {
  const grade = document.getElementById("gradeEcopontos");
  const filtrados = categoriaEscolhida === "Todos"
    ? LOCAIS_COLETA
    : LOCAIS_COLETA.filter(item => item.categoria === categoriaEscolhida);

  grade.innerHTML = filtrados.map(local => `
    <article class="cartao-ecoponto">
      <span class="tag-residuo">${local.categoria}</span>
      <h2>${local.nome}</h2>
      <p><strong>Bairro:</strong> ${local.bairro} · ${local.horario}</p>
      <a class="link-mapa" href="${local.mapaUrl}" target="_blank" rel="noopener">
        Abrir localizacao no Google Maps
      </a>
    </article>
  `).join("");
}

filtrarEcopontos("Todos");
\end{boxCodigoJS}

\section{Publicação no GitHub, Sala do Moodle e Plano B Offline}\label{sec:publicacao}

\passo{3.1}{Ativar o modo Template Repository no GitHub}
Suba a pasta para um repositório público no GitHub (ex.: \texttt{snct2026-starter-kit-curso1}). Na aba \botao{Settings} do repositório, marque a caixa \textbf{Template repository}. Assim, qualquer participante pode clicar no botão verde \botao{Use this template} ou baixar o \texttt{.zip}.

\passo{3.2}{Publicar o pacote \texttt{starter-kit-snct.zip} no Moodle institucional}
Com o papel de \textbf{Professor} na sala da oficina no Moodle do IFSC:
\begin{enumerate}[leftmargin=1.4em, itemsep=0.25em]
  \item Compacte a pasta do projeto como \texttt{starter-kit-snct.zip} (verifique se não há pasta oculta \texttt{.git} dentro do arquivo compactado);
  \item Na \textbf{Seção 1 — Bloco 1} do Moodle, acione \botao{Adicionar atividade ou recurso}\seta\botao{Arquivo} e envie o \texttt{starter-kit-snct.zip};
  \item Logo abaixo, adicione um recurso \botao{URL} apontando para o repositório no GitHub.
\end{enumerate}

\passo{3.3}{Preparar o Plano B Offline nos pen-drives da equipe}
Caso a internet do laboratório oscile no momento da oficina, grave em pelo menos \textbf{2 pen-drives}: (a) a pasta \texttt{starter-kit-snct/} já descompactada, (b) o arquivo \texttt{starter-kit-snct.zip} e (c) o PDF do Micro-Roteiro e os Slides-Guia da Frente 3.
